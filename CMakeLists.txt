cmake_minimum_required(VERSION 3.16)
project(kz_maptop_hud LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/deps/s2sdk" CACHE PATH "S2SDK CS2 checkout")
set(MMS "${CMAKE_CURRENT_SOURCE_DIR}/deps/metamod-source" CACHE PATH "Metamod API18 checkout")
option(MAPTOP_BUILD_PLUGIN "Build the Linux Metamod plugin" ON)
if(MSVC)
  add_compile_options(/W4 /utf-8 /UNDEBUG)
else()
  add_compile_options(-Wall -Wextra -UNDEBUG)
endif()
enable_testing()
add_executable(core_test tests/core_test.cpp)
target_include_directories(core_test PRIVATE src deps/json)
add_test(NAME core COMMAND core_test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME local_ranking COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/ranking_test.py $<TARGET_FILE:core_test>)

if(MAPTOP_BUILD_PLUGIN)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The server binary currently targets Linux x86_64. Build in Steam Runtime SDK Docker or WSL.")
  endif()
  if(NOT EXISTS "${MMS}/third_party/khook/include/khook.hpp")
    message(FATAL_ERROR "Run bootstrap.sh: Metamod API18 / KHook headers are required.")
  endif()
  file(READ "${SDK}/public/engine2/ivengineserver2.h" engine_header)
  if(NOT engine_header MATCHES "virtual bool[ \t]+IsSpawnGroupLoadedForClient")
    message(FATAL_ERROR "S2SDK predates the 2026-10-23 engine vtable update. Run bootstrap.sh.")
  endif()
  execute_process(COMMAND ${SDK}/devtools/bin/linux/protoc
    --proto_path=${SDK}/common --proto_path=${SDK}/thirdparty/protobuf-3.21.8/src
    --cpp_out=${CMAKE_CURRENT_BINARY_DIR} ${SDK}/common/network_connection.proto
    RESULT_VARIABLE proto_result)
  if(NOT proto_result EQUAL 0)
    message(FATAL_ERROR "SDK protoc failed: ${proto_result}")
  endif()
  add_library(kz_maptop_hud SHARED src/plugin.cpp ${CMAKE_CURRENT_BINARY_DIR}/network_connection.pb.cc
    ${SDK}/tier0_perproject/convar.cpp ${SDK}/public/tier0/memoverride.cpp)
  target_compile_definitions(kz_maptop_hud PRIVATE LINUX POSIX _LINUX GNUC COMPILER_GCC PLATFORM_64BITS
    _GLIBCXX_USE_CXX11_ABI=0 META_IS_SOURCE2 stricmp=strcasecmp strnicmp=strncasecmp)
  target_compile_options(kz_maptop_hud PRIVATE -fPIC -fvisibility=hidden -fno-gnu-unique -Wno-invalid-offsetof)
  target_include_directories(kz_maptop_hud PRIVATE src deps/json ${CMAKE_CURRENT_BINARY_DIR})
  target_include_directories(kz_maptop_hud SYSTEM PRIVATE deps/api ${MMS}/core ${MMS}/third_party/khook/include ${SDK}/public ${SDK}/public/tier0 ${SDK}/public/tier1
    ${SDK}/public/entity2 ${SDK}/public/interfaces ${SDK}/game/shared ${SDK}/common ${SDK}/public/mathlib
    ${SDK}/thirdparty/protobuf-3.21.8/src)
  target_link_libraries(kz_maptop_hud PRIVATE ${SDK}/lib/linux64/interfaces.a ${SDK}/lib/linux64/libtier0.so
    ${SDK}/lib/linux64/libsteam_api.so ${SDK}/lib/linux64/release/libprotobuf.a dl pthread)
  target_link_options(kz_maptop_hud PRIVATE -Wl,--exclude-libs,libprotobuf.a -Wl,-z,defs -Wl,-Bsymbolic)
  set_target_properties(kz_maptop_hud PROPERTIES PREFIX "" BUILD_WITH_INSTALL_RPATH TRUE
    INSTALL_RPATH "$ORIGIN/../../../../../bin/linuxsteamrt64")
  add_test(NAME binary_api18 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/binary_api_test.py $<TARGET_FILE:kz_maptop_hud>)
  set_tests_properties(binary_api18 PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${SDK}/lib/linux64")
  add_executable(config_test tests/config_test.cpp)
  target_include_directories(config_test PRIVATE src)
  target_include_directories(config_test SYSTEM PRIVATE $<TARGET_PROPERTY:kz_maptop_hud,INCLUDE_DIRECTORIES>)
  target_compile_definitions(config_test PRIVATE $<TARGET_PROPERTY:kz_maptop_hud,COMPILE_DEFINITIONS>)
  target_link_libraries(config_test PRIVATE ${SDK}/lib/linux64/libtier0.so)
  add_test(NAME cs2kz_config COMMAND config_test)
  set_tests_properties(cs2kz_config PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${SDK}/lib/linux64")
  foreach(production_test load_cvar menu)
    add_executable(${production_test}_test tests/${production_test}_test.cpp ${CMAKE_CURRENT_BINARY_DIR}/network_connection.pb.cc
      ${SDK}/tier0_perproject/convar.cpp ${SDK}/public/tier0/memoverride.cpp)
    target_include_directories(${production_test}_test SYSTEM PRIVATE $<TARGET_PROPERTY:kz_maptop_hud,INCLUDE_DIRECTORIES>)
    target_compile_definitions(${production_test}_test PRIVATE $<TARGET_PROPERTY:kz_maptop_hud,COMPILE_DEFINITIONS>)
    target_compile_options(${production_test}_test PRIVATE $<TARGET_PROPERTY:kz_maptop_hud,COMPILE_OPTIONS>)
    target_link_libraries(${production_test}_test PRIVATE $<TARGET_PROPERTY:kz_maptop_hud,LINK_LIBRARIES>)
    target_link_options(${production_test}_test PRIVATE $<TARGET_PROPERTY:kz_maptop_hud,LINK_OPTIONS>)
  endforeach()
  add_test(NAME load_cvar_tokenize COMMAND load_cvar_test)
  set_tests_properties(load_cvar_tokenize PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${SDK}/lib/linux64")
  add_test(NAME board_controls_actions COMMAND menu_test)
  set_tests_properties(board_controls_actions PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${SDK}/lib/linux64")
endif()
